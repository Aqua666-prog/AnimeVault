cmake_minimum_required(VERSION 3.22.1)
project(animevault_ai LANGUAGES C CXX ASM)
# Bound memory use in Ninja even on hosts exposing many logical CPUs.
set_property(GLOBAL PROPERTY JOB_POOLS mnn_compile=2 mnn_link=1)
set(CMAKE_JOB_POOL_COMPILE mnn_compile)
set(CMAKE_JOB_POOL_LINK mnn_link)

set(MNN_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../../third_party/MNN")
if(NOT EXISTS "${MNN_ROOT}/transformers/llm/engine/include/llm/llm.hpp")
    message(FATAL_ERROR "Run git submodule update --init third_party/MNN before building")
endif()
# Match the official Android LLM build, with network and multimodal features disabled.
# Command build avoids the Android header-export POST_BUILD on the LLM OBJECT target.
set(MNN_BUILD_FOR_ANDROID_COMMAND ON CACHE BOOL "" FORCE)
set(MNN_BUILD_LLM ON CACHE BOOL "" FORCE)
set(MNN_LOW_MEMORY ON CACHE BOOL "" FORCE)
set(MNN_CPU_WEIGHT_DEQUANT_GEMM ON CACHE BOOL "" FORCE)
set(MNN_SUPPORT_TRANSFORMER_FUSE ON CACHE BOOL "" FORCE)
set(MNN_ARM82 ON CACHE BOOL "" FORCE)
set(MNN_SEP_BUILD OFF CACHE BOOL "" FORCE)
set(MNN_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(MNN_BUILD_DEMO OFF CACHE BOOL "" FORCE)
set(MNN_LLM_BUILD_DEMO OFF CACHE BOOL "" FORCE)
set(MNN_BUILD_LLM_OMNI OFF CACHE BOOL "" FORCE)
set(LLM_SUPPORT_HTTP_RESOURCE OFF CACHE BOOL "" FORCE)
set(MNN_USE_LOGCAT ON CACHE BOOL "" FORCE)
set(CMAKE_SHARED_LINKER_FLAGS "${CMAKE_SHARED_LINKER_FLAGS} -Wl,-z,max-page-size=16384")
add_subdirectory("${MNN_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/mnn" EXCLUDE_FROM_ALL)
add_library(animevault_ai SHARED mnn_jni.cpp)
target_compile_features(animevault_ai PRIVATE cxx_std_17)
target_compile_options(animevault_ai PRIVATE -fexceptions -Wall -Wextra)
target_include_directories(animevault_ai PRIVATE
    "${MNN_ROOT}/include" "${MNN_ROOT}/transformers/llm/engine/include")
target_link_libraries(animevault_ai PRIVATE MNN log)
